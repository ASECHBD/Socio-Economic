  <script>
    let currentUser = null;
    let rawData = [];       
    let filteredData = [];  
    let currentPage = 1;
    const rowsPerPage = 10; 

    function showToast(msg, isSuccess = true) {
      const toast = $('#toast');
      toast.text(msg).removeClass('bg-red-600 bg-green-600')
           .addClass(isSuccess ? 'bg-green-600' : 'bg-red-600')
           .fadeIn().delay(3000).fadeOut();
    }

    function calculateAmount() {
      const inst = parseFloat($('#rec_installment').val()) || 0;
      const count = parseFloat($('#rec_installCount').val()) || 0;
      $('#rec_amount').val(inst * count);
    }

    function handleLogin() {
      const paId = $('#loginPaId').val();
      const pin = $('#loginPin').val();
      
      if (!paId || !pin) {
        showToast("PA ID এবং PIN দিন", false);
        return;
      }
      
      google.script.run.withSuccessHandler(res => {
        if (res.success) {
          currentUser = res.user;
          $('#loginSection').addClass('hidden');
          $('#appSection').removeClass('hidden');
          $('#userInfo').text(`User: ${currentUser.name} (ID: ${currentUser.id})`);
          applyPermissions();
          showToast("সফলভাবে লগইন হয়েছে!");
        } else {
          showToast(res.message, false);
        }
      }).checkLogin(paId, pin);
    }

    function handleLogout() {
      currentUser = null;
      $('#appSection').addClass('hidden');
      $('#loginSection').removeClass('hidden');
      $('#loginPaId').val('');
      $('#loginPin').val('');
    }

    function applyPermissions() {
      if (currentUser.permission === "1") {
        $('#nav-rfndSaving, #nav-profitPaid, #nav-dataReport').hide();
        switchTab('recSaving');
      } else {
        $('#nav-rfndSaving, #nav-profitPaid, #nav-dataReport').show();
        switchTab('recSaving');
      }
    }

    function switchTab(tabId) {
      $('.tab-content').addClass('hidden');
      $(`#tab-${tabId}`).removeClass('hidden');
      $('nav button').removeClass('bg-green-600').addClass('bg-green-800');
      $(`#nav-${tabId}`).removeClass('bg-green-800').addClass('bg-green-600');

      if (tabId === 'masterSearch') {
        loadAllMasterData();
      }
    }

    function fetchMSSData() {
      const mssNo = $('#rec_mssNo').val();
      if (!mssNo) return;
      
      google.script.run.withSuccessHandler(res => {
        if (res.success) {
          $('#rec_name').val(res.data.name);
          $('#rec_address').val(res.data.address);
          $('#rec_mobile').val(res.data.mobile);
          $('#rec_acOpenDate').val(res.data.acOpenDate);
          $('#rec_yrsMonths').val(res.data.yrsMonths);
          $('#rec_installment').val(res.data.installment);
          calculateAmount();
          showToast("S1MSSList থেকে ডাটা আনা হয়েছে!");
        } else {
          showToast(res.message, false);
        }
      }).getMSSDetails(mssNo);
    }

    function saveRecSaving(e) {
      e.preventDefault();
      const formData = {
        mssNo: $('#rec_mssNo').val(),
        name: $('#rec_name').val(),
        address: $('#rec_address').val(),
        mobile: $('#rec_mobile').val(),
        acOpenDate: $('#rec_acOpenDate').val(),
        yrsMonths: $('#rec_yrsMonths').val(),
        installment: $('#rec_installment').val(),
        installCount: $('#rec_installCount').val(),
        amount: $('#rec_amount').val(),
        remarks: $('#rec_remarks').val()
      };

      google.script.run.withSuccessHandler(res => {
        showToast(res.message);
        $('#recForm')[0].reset();
      }).submitRecSaving(formData, currentUser.id);
    }

    function saveRfndSaving(e) {
      e.preventDefault();
      const formData = {
        mssNo: $('#rfnd_mssNo').val(),
        name: $('#rfnd_name').val(),
        principal: $('#rfnd_principal').val()
      };

      google.script.run.withSuccessHandler(res => {
        showToast(res.message);
        $('#rfndForm')[0].reset();
      }).submitRfndSaving(formData, currentUser.id);
    }

    function saveProfit(e) {
      e.preventDefault();
      const formData = {
        mssNo: $('#profit_mssNo').val(),
        name: $('#profit_name').val(),
        profit: $('#profit_amount').val()
      };

      google.script.run.withSuccessHandler(res => {
        showToast(res.message);
        $('#profitForm')[0].reset();
      }).submitProfit(formData, currentUser.id);
    }

    // ================= MASTER SEARCH PAGINATION LOGIC =================
    function loadAllMasterData() {
      document.getElementById("tableBody").innerHTML = '<tr><td colspan="8" class="text-center py-4">ডেটা লোড হচ্ছে...</td></tr>';
      
      google.script.run
        .withSuccessHandler(function(data) {
          rawData = data || [];
          filteredData = rawData;
          currentPage = 1;
          renderTable();
        })
        .withFailureHandler(function(err) {
          document.getElementById("tableBody").innerHTML = '<tr><td colspan="8" class="text-center py-4 text-red-600">ডেটা লোড করতে সমস্যা হয়েছে!</td></tr>';
        })
        .getAllS5MasterData();
    }

    function filterData() {
      const query = document.getElementById("searchInput").value.toLowerCase().trim();
      
      filteredData = rawData.filter(item => {
        return (item.mssNo && item.mssNo.toString().toLowerCase().includes(query)) || 
               (item.name && item.name.toString().toLowerCase().includes(query));
      });

      currentPage = 1; 
      renderTable();
    }

    function renderTable() {
      const tbody = document.getElementById("tableBody");
      tbody.innerHTML = "";

      if (filteredData.length === 0) {
        tbody.innerHTML = '<tr><td colspan="8" class="text-center py-4">কোনো তথ্য পাওয়া যায়নি।</td></tr>';
        updatePaginationControls(0);
        return;
      }

      const start = (currentPage - 1) * rowsPerPage;
      const end = start + rowsPerPage;
      const paginatedItems = filteredData.slice(start, end);

      paginatedItems.forEach(item => {
        const tr = document.createElement("tr");
        tr.innerHTML = `
          <td>${item.mssNo || ''}</td>
          <td>${item.name || ''}</td>
          <td>${item.acOpenDt || ''}</td>
          <td>${item.yrs || ''}</td>
          <td>${item.installment || ''}</td>
          <td>${item.monthKisti || 0}</td>
          <td>${item.principal || 0}</td>
          <td>${item.remarks || ''}</td>
        `;
        tbody.appendChild(tr);
      });

      updatePaginationControls(filteredData.length);
    }

    function updatePaginationControls(totalItems) {
      const totalPages = Math.ceil(totalItems / rowsPerPage) || 1;
      document.getElementById("pageInfo").innerText = `Page ${currentPage} of ${totalPages}`;

      document.getElementById("btnPrev").disabled = (currentPage === 1);
      document.getElementById("btnNext").disabled = (currentPage === totalPages);
    }

    function prevPage() {
      if (currentPage > 1) {
        currentPage--;
        renderTable();
      }
    }

    function nextPage() {
      const totalPages = Math.ceil(filteredData.length / rowsPerPage);
      if (currentPage < totalPages) {
        currentPage++;
        renderTable();
      }
    }

    // ================= DATE RANGE REPORT =================
    function generateReport() {
      const sheet = $('#reportSheetName').val();
      const fromDate = $('#reportFromDate').val();
      const toDate = $('#reportToDate').val();

      if (!fromDate || !toDate) return showToast("তারিখ সিলেক্ট করুন", false);

      google.script.run.withSuccessHandler(data => {
        const wrapper = $('#reportTableWrapper');
        wrapper.empty(); // Clean up previous table

        if (!data || !data.headers || data.rows.length === 0) {
          showToast("কোনো তথ্য পাওয়া যায়নি", false);
          return;
        }

        // Rebuild Table HTML
        let tableHtml = '<table id="reportTable" class="display w-full"><thead><tr>';
        data.headers.forEach(h => tableHtml += `<th>${h}</th>`);
        tableHtml += '</tr></thead><tbody>';

        data.rows.forEach(r => {
          tableHtml += '<tr>';
          data.headers.forEach(h => {
            tableHtml += `<td>${r[h] || ''}</td>`;
          });
          tableHtml += '</tr>';
        });
        tableHtml += 'tbody></table>';

        wrapper.html(tableHtml);
        $('#reportTable').DataTable({ pageLength: 10 });
        showToast("রিপোর্ট লোড হয়েছে!");
      }).getDateRangeReport(sheet, fromDate, toDate);
    }
  </script>
