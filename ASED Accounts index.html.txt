<script>
let appData = {};

function login() {
  const id = $('#paId').val();
  const pin = $('#paPin').val();
  if(!id || !pin) return;
  $('#loader').show();
  google.script.run.withSuccessHandler(res => {
    $('#loader').hide();
    if(res.success) {
      $('#loginPage').hide(); $('#mainApp').show();
      init();
      const Toast = Swal.mixin({toast: true, position: 'top-end', showConfirmButton: false, timer: 3000});
      Toast.fire({icon: 'success', title: 'স্বাগতম, ' + res.user});
    } else { Swal.fire('Error', res.msg, 'error'); }
  }).checkLogin(id, pin);
}

function init() {
  google.script.run.withSuccessHandler(data => {
    appData = data;
    let opt = '<option value="">নির্বাচন করুন</option>';
    data.mssList.forEach(m => opt += `<option value="${m.id}">${m.id}</option>`);
    $('.mss-drop').html(opt);
    renderTable(data.masterTable);
  }).getInitialData();
}

$(document).on('change', '.mss-drop', function() {
  const m = appData.mssList.find(x => x.id == $(this).val());
  const f = $(this).closest('form');
  if(m) { f.find('.n-field').val(m.name); f.find('.i-field').val(m.inst); }
});

function handleForm(id, type) {
  $(`#${id}`).submit(function(e) {
    e.preventDefault();
    $('#loader').show();
    const obj = {};
    $(this).serializeArray().forEach(i => obj[i.name] = i.value);
    google.script.run.withSuccessHandler(msg => {
      $('#loader').hide();
      Swal.fire('সফল', msg, 'success');
      $(`#${id}`)[0].reset();
      init();
    }).processEntry(type, obj);
  });
}

handleForm('fRec', 'rec'); handleForm('fRef', 'ref'); handleForm('fProf', 'profit');

function renderTable(data) {
  if ($.fn.DataTable.isDataTable('#mTable')) $('#mTable').DataTable().destroy();
  let html = '';
  data.forEach(r => {
    html += `<tr><td>${r[0]}</td><td>${r[1]}</td><td>${r[6]}</td><td>${r[7]}</td><td>${r[10]}</td><td>${r[14] || ''}</td></tr>`;
  });
  $('#mBody').html(html);
  $('#mTable').DataTable({ pageLength: 10, language: { search: "খুঁজুন:" } });
}
