  <script src="https://cdn.jsdelivr.net/npm/bootstrap@5.3.0/dist/js/bootstrap.bundle.min.js"></script>
  <script>
    let basics = [];

    function login() {
      const id = document.getElementById('loginPaId').value;
      const pin = document.getElementById('loginPin').value;
      google.script.run.withSuccessHandler(res => {
        if(res.status) {
          document.getElementById('login-box').style.display='none';
          document.getElementById('main-form').style.display='block';
          document.getElementById('paId').value = id;
          document.getElementById('paName').innerText = res.name;
          loadData();
        } else { document.getElementById('loginError').style.display='block'; }
      }).verifyUser(id, pin);
    }

    // লগআউট ফাংশন
    function logout() {
      // ফর্ম রিফ্রেশ এবং লগইন বক্সে ফিরে যাওয়া
      document.getElementById('main-form').style.display='none';
      document.getElementById('login-box').style.display='block';
      document.getElementById('loginPaId').value = "";
      document.getElementById('loginPin').value = "";
      document.getElementById('loginError').style.display='none';
      
      // Toast মেসেজ দেখানো
      const toastElement = document.getElementById('logoutToast');
      const toast = new bootstrap.Toast(toastElement);
      toast.show();
    }

    function loadData() {
      google.script.run.withSuccessHandler(n => { 
        basics = n; 
        document.getElementById('rows').innerHTML = ""; // আগের রো ক্লিয়ার করা
        for(let i=0; i<10; i++) addRow(i); 
      }).getBasicNames();
      document.getElementById('date').valueAsDate = new Date();
    }

    function addRow(i) {
      const html = `<tr>
        <td><select id="b_${i}" class="form-select" onchange="loadP(${i})"><option value="">Select</option>${basics.map(x=>`<option>${x}</option>`).join('')}</select></td>
        <td><select id="p_${i}" class="form-select" onchange="setD(${i})"><option value="">Select</option></select></td>
        <td><input type="text" id="c_${i}" class="form-control" readonly></td>
        <td><input type="text" id="v_${i}" class="form-control" readonly></td>
        <td><input type="number" id="q_${i}" class="form-control" oninput="calc(${i})"></td>
        <td><input type="text" id="a_${i}" class="form-control total-cell" readonly></td>
      </tr>`;
      document.getElementById('rows').insertAdjacentHTML('beforeend', html);
    }

    function loadP(i) {
      const b = document.getElementById(`b_${i}`).value;
      google.script.run.withSuccessHandler(p => {
        document.getElementById(`p_${i}`).innerHTML = '<option value="">Select</option>' + p.map(x=>`<option value='${JSON.stringify(x)}'>${x.name}</option>`).join('');
      }).getProductsByBasic(b);
    }

    function setD(i) {
      const data = document.getElementById(`p_${i}`).value;
      if(!data) return;
      const p = JSON.parse(data);
      document.getElementById(`c_${i}`).value = p.code;
      document.getElementById(`v_${i}`).value = p.value;
      calc(i);
    }

    function calc(i) {
      const v = Number(document.getElementById(`v_${i}`).value);
      const q = Number(document.getElementById(`q_${i}`).value);
      document.getElementById(`a_${i}`).value = v * q;
      let total = 0;
      document.querySelectorAll('.total-cell').forEach(el => total += Number(el.value));
      document.getElementById('grandTotal').innerText = total;
    }

    function submitData() {
      const btn = document.getElementById('subBtn');
      btn.disabled = true; btn.innerText = "Processing...";
      const prods = [];
      for(let i=0; i<10; i++) {
        const pSel = document.getElementById(`p_${i}`).value;
        if(pSel) {
          prods.push({
            productName: JSON.parse(pSel).name,
            code: document.getElementById(`c_${i}`).value,
            value: document.getElementById(`v_${i}`).value,
            qty: document.getElementById(`q_${i}`).value,
            amount: document.getElementById(`a_${i}`).value
          });
        }
      }
      const data = {
        date: document.getElementById('date').value,
        paId: document.getElementById('paId').value,
        beneficiaryName: document.getElementById('bName').value,
        mobile: document.getElementById('mobile').value,
        address: document.getElementById('address').value,
        total: document.getElementById('grandTotal').innerText,
        deliveryComment: document.getElementById('comment').value,
        products: prods
      };
      google.script.run.withSuccessHandler(() => { alert("Success!"); location.reload(); }).submitOrder(data);
    }
