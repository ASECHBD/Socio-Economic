<script src="https://cdn.jsdelivr.net/npm/bootstrap@5.3.0/dist/js/bootstrap.bundle.min.js"></script>
<script>
  let products = [], stockData = [];

  function login() {
    const id = document.getElementById('l_id').value;
    const pin = document.getElementById('l_pin').value;
    google.script.run.withSuccessHandler(res => {
      if(res.status) { document.getElementById('login-box').style.display='none'; document.getElementById('main-app').style.display='block'; init(); }
      else alert("Login Failed!");
    }).verifyUser(id, pin);
  }

  function init() {
    google.script.run.withSuccessHandler(d => { products = d; addRow('rec'); addRow('issue'); addRow('adj'); }).getProductData();
    refreshStock();
  }

  function addRow(type) {
    const container = document.getElementById(type==='issue'?'iss-rows':(type==='rec'?'rec-rows':'adj-rows'));
    const id = Date.now();
    const bOpts = [...new Set(products.map(p => p.basic))].map(b => `<option value="${b}">${b}</option>`).join('');

    let html = `<div class="row g-2 item-row align-items-end" id="row-${id}">
      <div class="col-md-2"><label>Basic</label><select class="form-select form-select-sm" onchange="onBChange('${type}',${id},this.value)"><option value="">Select</option>${bOpts}</select></div>
      <div class="col-md-3"><label>Product</label><select class="form-select form-select-sm p-name" id="p-${id}" onchange="onPChange('${type}',${id},this.value)"><option value="">Select</option></select></div>
      <div class="col-md-2"><label>Code</label><input type="text" class="form-control form-control-sm p-code" id="c-${id}" readonly></div>`;

    if(type === 'issue') {
      html += `<div class="col-md-1"><label>Price</label><input type="number" class="form-control form-control-sm" id="pr-${id}" readonly></div>
               <div class="col-md-1"><label>Qty</label><input type="number" class="form-control form-control-sm p-qty" id="q-${id}" oninput="calcRow(${id})"></div>
               <div class="col-md-2"><label>Amount</label><input type="text" class="form-control form-control-sm p-amt" id="a-${id}" readonly></div>`;
    } else if(type === 'rec') {
      html += `<div class="col-md-4"><label>Qty</label><input type="number" class="form-control form-control-sm p-qty"></div>`;
    } else {
      html += `<div class="col-md-2"><label>(+)Plus</label><input type="number" class="form-control form-control-sm p-plus" value="0"></div>
               <div class="col-md-2"><label>(-)Minus</label><input type="number" class="form-control form-control-sm p-minus" value="0"></div>`;
    }
    html += `<div class="col-md-1 text-center"><span class="btn-remove" onclick="document.getElementById('row-${id}').remove(); updateGrandTotal();">×</span></div></div>`;
    container.insertAdjacentHTML('beforeend', html);
  }

  function onBChange(t,id,v) {
    const f = products.filter(p => p.basic === v);
    document.getElementById(`p-${id}`).innerHTML = '<option value="">Select</option>' + f.map(p => `<option value="${p.name}">${p.name}</option>`).join('');
  }

  function onPChange(t,id,v) {
    const item = products.find(p => p.name === v);
    if(item) {
      document.getElementById(`c-${id}`).value = item.code;
      if(t === 'issue') { document.getElementById(`pr-${id}`).value = item.price; calcRow(id); }
    }
  }

  function calcRow(id) {
    const p = parseFloat(document.getElementById(`pr-${id}`).value) || 0;
    const q = parseFloat(document.getElementById(`q-${id}`).value) || 0;
    document.getElementById(`a-${id}`).value = (p * q).toFixed(2);
    updateGrandTotal();
  }

  function updateGrandTotal() {
    let g = 0; document.querySelectorAll('.p-amt').forEach(el => g += parseFloat(el.value) || 0);
    document.getElementById('iss_grand_total').innerText = g.toFixed(2);
  }

  function submitData(type) {
    const btn = document.getElementById('btn-'+type);
    const container = document.getElementById(type==='issue'?'iss-rows':(type==='rec'?'rec-rows':'adj-rows'));
    const rows = Array.from(container.getElementsByClassName('item-row'));
    if(rows.length === 0) return alert("Items missing!");
    
    btn.disabled = true; btn.innerText = "Saving...";
    const items = rows.map(row => {
      return {
        bName: row.querySelector('select').value,
        pName: row.querySelector('.p-name').value,
        pCode: row.querySelector('.p-code').value,
        qty: row.querySelector('.p-qty')?.value || 0,
        price: row.querySelector('[id^="pr-"]')?.value || 0,
        total: row.querySelector('.p-amt')?.value || 0,
        plusQty: row.querySelector('.p-plus')?.value || 0,
        minusQty: row.querySelector('.p-minus')?.value || 0
      };
    });

    const common = { 
      chNo: v('rec_ch'), supplier: v('rec_sup'), mobile: v('rec_mob')||v('iss_mob'), 
      billNo: v('iss_bill'), benName: v('iss_ben'), addr: v('iss_addr'), cause: v('adj_cause') 
    };

    google.script.run.withSuccessHandler(() => {
      document.getElementById('toast').style.display='block';
      container.innerHTML=''; addRow(type); refreshStock();
      btn.disabled = false; btn.innerText = "SAVED";
      setTimeout(() => { document.getElementById('toast').style.display='none'; btn.innerText = "SAVE "+type.toUpperCase(); }, 2500);
    }).saveData(type, common, items);
  }

  function fetchReport() {
    const type = document.getElementById('rep_type').value;
    const start = document.getElementById('rep_start').value;
    const end = document.getElementById('rep_end').value;
    const btn = document.getElementById('btn-search');
    
    btn.disabled = true; btn.innerText = "Searching...";
    google.script.run.withSuccessHandler(res => {
      btn.disabled = false; btn.innerText = "SEARCH REPORT";
      if (res.error) return alert(res.error);

      document.getElementById('rep_head').innerHTML = "<tr>" + res.headers.map(h => `<th>${h}</th>`).join('') + "</tr>";
      let b = "";
      if (res.rows.length === 0) b = `<tr><td colspan="${res.headers.length}" class="text-center">No results for this date/time range.</td></tr>`;
      else res.rows.forEach(r => b += "<tr>" + r.map(c => `<td>${c||""}</td>`).join('') + "</tr>");
      document.getElementById('rep_body').innerHTML = b;
    }).searchReports(type, start, end);
  }

  function refreshStock() { google.script.run.withSuccessHandler(d => renderStock(d)).getMasterStock(); }
  function renderStock(d) {
    let h = ''; 
    d.forEach(r => { 
      const s = parseFloat(r[10]) || 0; 
      const cls = s <= 0 ? 'table-danger' : (s <= 5 ? 'table-warning' : '');
      h += `<tr class="${cls}"><td>${r[0]}</td><td>${r[1]}</td><td>${r[2]}</td><td>${s}</td></tr>`; 
    });
    document.getElementById('stock-list').innerHTML = h;
  }

  function filterStock() {
    const q = document.getElementById('stockSearch').value.toLowerCase();
    document.querySelectorAll('#stock-list tr').forEach(tr => tr.style.display = tr.innerText.toLowerCase().includes(q) ? '' : 'none');
  }

  function v(id) { const el = document.getElementById(id); return el ? el.value : ''; }
  function logout() { location.reload(); }
</script>
